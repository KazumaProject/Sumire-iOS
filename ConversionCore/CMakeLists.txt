cmake_minimum_required(VERSION 3.20)
project(SumireConversion LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ZLIB REQUIRED)
add_library(sumire_conversion src/engine.cpp src/c_api.cpp src/unicode_case.cpp src/dictionary_browser.cpp)
target_include_directories(sumire_conversion PUBLIC src)
target_link_libraries(sumire_conversion PRIVATE ZLIB::ZLIB)
if(APPLE)
  target_link_libraries(sumire_conversion PRIVATE "-framework CoreFoundation")
else()
  find_package(ICU REQUIRED COMPONENTS uc)
  target_link_libraries(sumire_conversion PRIVATE ICU::uc)
endif()
target_compile_options(sumire_conversion PRIVATE -Wall -Wextra -Wpedantic)
add_executable(sumire-cli src/main.cpp)
target_link_libraries(sumire-cli PRIVATE sumire_conversion)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME conversion COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:sumire-cli> $<TARGET_FILE:sumire-c-api-test>)

add_executable(sumire-c-api-test tests/test_c_api.c)
target_link_libraries(sumire-c-api-test PRIVATE sumire_conversion)

add_executable(sumire-bunsetsu-test tests/test_bunsetsu.cpp)
target_link_libraries(sumire-bunsetsu-test PRIVATE sumire_conversion)
add_test(NAME bunsetsu COMMAND sumire-bunsetsu-test)

add_executable(sumire-browser-test tests/test_dictionary_browser.cpp)
target_link_libraries(sumire-browser-test PRIVATE sumire_conversion)
add_test(NAME dictionary_browser COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_dictionary_browser.py $<TARGET_FILE:sumire-browser-test>)
