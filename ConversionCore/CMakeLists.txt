cmake_minimum_required(VERSION 3.20)
project(SumireConversion LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ZLIB REQUIRED)
add_library(sumire_conversion src/engine.cpp src/c_api.cpp src/unicode_case.cpp src/dictionary_browser.cpp src/system_ngram.cpp src/mozc_scoring_data.cpp)
target_include_directories(sumire_conversion PUBLIC src)
target_link_libraries(sumire_conversion PRIVATE ZLIB::ZLIB)
if(APPLE)
  target_link_libraries(sumire_conversion PRIVATE "-framework CoreFoundation")
else()
  find_package(ICU REQUIRED COMPONENTS uc)
  target_link_libraries(sumire_conversion PRIVATE ICU::uc)
endif()
target_compile_options(sumire_conversion PRIVATE -Wall -Wextra -Wpedantic)
add_executable(sumire-cli src/main.cpp)
target_link_libraries(sumire-cli PRIVATE sumire_conversion)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME standalone_builder COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../Scripts/tests/test_standalone_dictionary.py)
add_test(NAME conversion COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:sumire-cli> $<TARGET_FILE:sumire-c-api-test> $<TARGET_FILE:sumire-incremental-test>)
add_test(NAME reference_quality_policy COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_reference_quality_policy.py)
add_test(NAME prediction_quality_policy COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_prediction_reference_policy.py)
add_test(NAME android_display_export_policy COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_android_display_export.py)
add_test(NAME display_reference_comparison COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_display_reference_comparison.py)
add_test(NAME authored_conversion_quality COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_conversion_quality.py)

add_executable(sumire-c-api-test tests/test_c_api.c)
target_link_libraries(sumire-c-api-test PRIVATE sumire_conversion)

add_executable(sumire-bunsetsu-test tests/test_bunsetsu.cpp)
target_link_libraries(sumire-bunsetsu-test PRIVATE sumire_conversion)
add_test(NAME bunsetsu COMMAND sumire-bunsetsu-test)
add_executable(sumire-system-ngram-test tests/test_system_ngram.cpp)
target_link_libraries(sumire-system-ngram-test PRIVATE sumire_conversion)
add_test(NAME system_ngram COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_system_ngram.py $<TARGET_FILE:sumire-system-ngram-test>)
add_executable(sumire-mozc-scoring-test tests/test_mozc_scoring_data.cpp)
target_link_libraries(sumire-mozc-scoring-test PRIVATE sumire_conversion)
add_test(NAME mozc_scoring_data COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_mozc_scoring_data.py $<TARGET_FILE:sumire-mozc-scoring-test>)

add_executable(sumire-browser-test tests/test_dictionary_browser.cpp)
target_link_libraries(sumire-browser-test PRIVATE sumire_conversion)
add_test(NAME dictionary_browser COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_dictionary_browser.py $<TARGET_FILE:sumire-browser-test>)
add_executable(sumire-incremental-test tests/test_incremental.cpp)
target_link_libraries(sumire-incremental-test PRIVATE sumire_conversion)
